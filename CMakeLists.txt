cmake_minimum_required(VERSION 3.24)
project(digital_globe VERSION 0.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(DIGITAL_GLOBE_BUILD_TESTS "Build regression tests" ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()

include(cmake/WxWidgets.cmake)
find_package(OpenGL REQUIRED)
find_package(Threads REQUIRED)
if(WIN32)
  include(FetchContent)
  set(BUILD_CURL_EXE OFF CACHE BOOL "" FORCE)
  set(BUILD_LIBCURL_DOCS OFF CACHE BOOL "" FORCE)
  set(BUILD_MISC_DOCS OFF CACHE BOOL "" FORCE)
  set(ENABLE_CURL_MANUAL OFF CACHE BOOL "" FORCE)
  set(CURL_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  set(HTTP_ONLY ON CACHE BOOL "" FORCE)
  set(CURL_USE_SCHANNEL ON CACHE BOOL "" FORCE)
  set(CURL_USE_OPENSSL OFF CACHE BOOL "" FORCE)
  set(CURL_USE_LIBPSL OFF CACHE BOOL "" FORCE)
  set(CURL_ZLIB OFF CACHE BOOL "" FORCE)
  if(DIGITAL_GLOBE_WX_PROVIDER STREQUAL "BUNDLED")
    set(CURL_STATIC_CRT ON CACHE BOOL "" FORCE)
  else()
    set(CURL_STATIC_CRT OFF CACHE BOOL "" FORCE)
  endif()
  FetchContent_Declare(curl URL https://curl.se/download/curl-8.23.0.tar.xz
    URL_HASH SHA256=8dadd7d2536a314f982ad05a651b713f426bf7d02cc2c76cffb9796f2218e6c5 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(curl)
else()
  find_package(CURL REQUIRED)
endif()

add_library(globe_core STATIC de2.cpp sky.cpp util.cpp spheroid.cpp app_settings.cpp platform.cpp http_client.cpp)
target_include_directories(globe_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(globe_core PUBLIC ${DIGITAL_GLOBE_WX_LIBRARIES} OpenGL::GL Threads::Threads CURL::libcurl ${CMAKE_DL_LIBS})
target_compile_definitions(globe_core PUBLIC GLM_ENABLE_EXPERIMENTAL)
if(WIN32)
  target_compile_definitions(globe_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS)
endif()

add_executable(digital_globe WIN32 main.cpp app_menu.cpp)
target_link_libraries(digital_globe PRIVATE globe_core)
if(WIN32)
  target_sources(digital_globe PRIVATE app.rc)
endif()
add_custom_command(TARGET digital_globe POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/shaders" "$<TARGET_FILE_DIR:digital_globe>/shaders"
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:digital_globe>/assets")

if(WIN32 AND DIGITAL_GLOBE_WX_PROVIDER STREQUAL "SYSTEM")
  get_filename_component(wx_dll_dir "${wxWidgets_LIB_DIR}" ABSOLUTE)
  if(wxWidgets_CONFIGURATION MATCHES "d$")
    file(GLOB wx_runtime_dlls "${wx_dll_dir}/wx*ud_*.dll")
    file(GLOB wx_gl_runtime "${wx_dll_dir}/wx*ud_gl_*.dll")
  else()
    file(GLOB wx_runtime_dlls "${wx_dll_dir}/wx*u_*.dll")
    file(GLOB wx_gl_runtime "${wx_dll_dir}/wx*u_gl_*.dll")
  endif()
  if(NOT wx_gl_runtime)
    message(FATAL_ERROR "The selected wxWidgets SDK is missing its matching OpenGL runtime DLL in ${wx_dll_dir}. Install or rebuild the SDK's gl component for ${wxWidgets_CONFIGURATION}.")
  endif()
  add_custom_target(stage_wx_runtime
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:digital_globe>"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different ${wx_runtime_dlls} "$<TARGET_FILE_DIR:digital_globe>"
    DEPENDS ${wx_runtime_dlls})
  add_dependencies(digital_globe stage_wx_runtime)
endif()

if(DIGITAL_GLOBE_BUILD_TESTS)
  enable_testing()
  foreach(test globe_tests render_tests visibility_tests cache_tests status_bar_tests menu_tests)
    add_executable(${test} "tests/${test}.cpp")
    target_link_libraries(${test} PRIVATE globe_core)
    if(TARGET stage_wx_runtime)
      add_dependencies(${test} stage_wx_runtime)
    endif()
    if(test STREQUAL "menu_tests")
      target_sources(${test} PRIVATE app_menu.cpp)
    endif()
    add_test(NAME ${test} COMMAND ${test})
    set_tests_properties(${test} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" TIMEOUT 120)
  endforeach()
  add_test(NAME application_smoke COMMAND digital_globe --smoke-test)
  set_tests_properties(application_smoke PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}" TIMEOUT 60)
endif()

include(GNUInstallDirs)
install(TARGETS digital_globe RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(WIN32)
  install(DIRECTORY shaders assets DESTINATION ${CMAKE_INSTALL_BINDIR})
  if(wx_runtime_dlls)
    install(FILES ${wx_runtime_dlls} DESTINATION ${CMAKE_INSTALL_BINDIR})
  endif()
else()
  install(DIRECTORY shaders assets DESTINATION ${CMAKE_INSTALL_DATADIR}/digital_globe)
endif()
install(FILES LICENSE DEPENDENCIES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/digital_globe)

if(WIN32 AND EXISTS "${curl_SOURCE_DIR}/COPYING")
  install(FILES "${curl_SOURCE_DIR}/COPYING" DESTINATION ${CMAKE_INSTALL_DATADIR}/digital_globe/licenses RENAME libcurl.txt)
endif()
if(EXISTS "${DIGITAL_GLOBE_WX_LICENSE}")
  install(FILES "${DIGITAL_GLOBE_WX_LICENSE}" DESTINATION ${CMAKE_INSTALL_DATADIR}/digital_globe/licenses RENAME wxWidgets.txt)
endif()
if(UNIX AND NOT APPLE)
  install(FILES packaging/digital_globe.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
endif()
set(CPACK_PACKAGE_NAME "DigitalGlobe")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
if(WIN32)
  set(CPACK_GENERATOR "ZIP")
else()
  set(CPACK_GENERATOR "TGZ")
endif()
include(CPack)
